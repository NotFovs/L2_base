// 1. hasTwoCubeSums
const hasTwoCubeSums = n => {
  let count = 0;
  for (let a = 1; a ** 3 < n; a++) {
    const rem = n - a ** 3;
    const b = Math.round(Math.cbrt(rem));
    if (b ** 3 === rem && b > a && ++count === 2) return true;
  }
  return false;
};

// 3. whatCentury
const whatCentury = year => {
  const c = Math.ceil(year / 100);
  const s = c % 100, d = c % 10;
  return c + (s >= 11 && s <= 13 ? 'th' : { 1: 'st', 2: 'nd', 3: 'rd' }[d] || 'th');
};

// 5. primeFactors
const primeFactors = n => {
  let res = '';
  for (let i = 2; i * i <= n; i++) {
    let c = 0;
    while (n % i === 0) { c++; n /= i; }
    if (c) res += c === 1 ? `(${i})` : `(${i}**${c})`;
  }
  return res + (n > 1 ? `(${n})` : '');
};

// 6. toWeirdCase
const toWeirdCase = s => s.replace(/\w+/g, w =>
  [...w].map((c, i) => i % 2 ? c.toLowerCase() : c.toUpperCase()).join('')
);

// 7. wave
const wave = str => [...str].reduce((acc, c, i) => {
  if (c !== ' ') acc.push(str.slice(0, i) + c.toUpperCase() + str.slice(i + 1));
  return acc;
}, []);

// 8. expandedForm
const expandedForm = n => [...String(n)]
  .map((d, i, a) => d * 10 ** (a.length - 1 - i))
  .filter(Boolean)
  .join(' + ');

// 9. solution (разбить на пары)
const solution = s => (s + '_'.repeat(s.length % 2)).match(/../g) || [];

// 10. bingo
const bingo = (ticket, win) =>
  ticket.filter(([w, num]) => [...w].some(c => c.charCodeAt(0) === num)).length >= win
    ? 'Winner!' : 'Loser!';

// 11. domainName
function domainName(url) {
  const u = new URL(url.includes('://') ? url : `http://${url}`);
  return u.hostname.replace(/^www\./, '').split('.')[0];
}
