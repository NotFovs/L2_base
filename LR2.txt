// 1. hasTwoCubeSums
const hasTwoCubeSums = n => {
  let count = 0;
  for (let a = 1; a ** 3 < n; a++) {
    const rem = n - a ** 3;
    const b = Math.round(Math.cbrt(rem));
    if (b ** 3 === rem && b > a && ++count === 2) return true;
  }
  return false;
};

// 3. whatCentury
const whatCentury = year => {
  const c = Math.ceil(year / 100);
  const s = c % 100, d = c % 10;
  return c + (s >= 11 && s <= 13 ? 'th' : { 1: 'st', 2: 'nd', 3: 'rd' }[d] || 'th');
};

// 5. primeFactors
const primeFactors = n => {
  let res = '';
  for (let i = 2; i * i <= n; i++) {
    let c = 0;
    while (n % i === 0) { c++; n /= i; }
    if (c) res += c === 1 ? `(${i})` : `(${i}**${c})`;
  }
  return res + (n > 1 ? `(${n})` : '');
};

// 6. toWeirdCase
const toWeirdCase = s => s.replace(/\w+/g, w =>
  [...w].map((c, i) => i % 2 ? c.toLowerCase() : c.toUpperCase()).join('')
);

// 7. wave
const wave = str => [...str].reduce((acc, c, i) => {
  if (c !== ' ') acc.push(str.slice(0, i) + c.toUpperCase() + str.slice(i + 1));
  return acc;
}, []);

// 8. expandedForm
const expandedForm = n => [...String(n)]
  .map((d, i, a) => d * 10 ** (a.length - 1 - i))
  .filter(Boolean)
  .join(' + ');

// 9. solution (разбить на пары)
const solution = s => (s + '_'.repeat(s.length % 2)).match(/../g) || [];

// 10. bingo
const bingo = (ticket, win) =>
  ticket.filter(([w, num]) => [...w].some(c => c.charCodeAt(0) === num)).length >= win
    ? 'Winner!' : 'Loser!';

// 11. domainName
const domainName = url => url.replace(/^https?:\/\/(www\.)?/, '').split('.')[0];

// 12. longest
const longest = (arr, n) =>
  [...arr].sort((a, b) => b.length - a.length || arr.indexOf(a) - arr.indexOf(b))[n - 1];

// 13. hexStringToRGB
const hexStringToRGB = s => ({
  r: parseInt(s.slice(1, 3), 16),
  g: parseInt(s.slice(3, 5), 16),
  b: parseInt(s.slice(5, 7), 16)
});

// 14. isCircleSorted
const isCircleSorted = arr =>
  arr.filter((x, i) => x > arr[(i + 1) % arr.length]).length <= 1;

// 15-16. Методы массива
Object.assign(Array.prototype, {
  square() { return this.map(x => x * x); },
  cube() { return this.map(x => x ** 3); },
  sum() { return this.reduce((a, b) => a + b, 0); },
  average() { return this.length ? this.sum() / this.length : NaN; },
  even() { return this.filter(x => !(x % 2)); },
  odd() { return this.filter(x => x % 2); },
});

// 18. createPhoneNumber
const createPhoneNumber = n =>
  `(${n.slice(0, 3).join('')}) ${n.slice(3, 6).join('')}-${n.slice(6).join('')}`;
